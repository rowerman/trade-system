#### 1.3.5 公共成员函数

##### 单例访问函数
```cpp
static CConfig& instance();  // 获取唯一实例的静态方法
```

**实现（config.cpp 第48-56行）：**
```cpp
CConfig& CConfig::instance() {
    if (pinstance_ == nullptr) {                    // 第一次检查（快速路径）
        std::lock_guard<mutex> g(instancelock_);    // 加锁
        if (pinstance_ == nullptr) {                // 第二次检查（双重检查锁定）
            pinstance_ = new CConfig();             // 创建唯一实例
        }
    }
    return *pinstance_;                              // 返回引用
}
```

**详细说明：**
- 使用**双重检查锁定（Double-Checked Locking）**模式
- 第一次检查：如果实例已存在，直接返回（避免不必要的加锁）
- 加锁：保护临界区
- 第二次检查：防止多线程同时通过第一次检查后重复创建
- 返回引用：避免返回指针可能为空的问题

```cpp
string CConfig::CtpSymbolToSecurityFullName(const std::string& symbol) {
    string fullsymbol;
    string num = symbol;                                    // 复制符号
    string alpha = symbol;                                  // 复制符号
    num.erase(std::remove_if(num.begin(), num.end(), &::isalpha), num.end());      // 移除字母，保留数字
    alpha.erase(std::remove_if(alpha.begin(), alpha.end(), &::isdigit), alpha.end()); // 移除数字，保留字母
    string tmp = instrument2sec[alpha];                     // 从映射中查找完整代码前缀
    fullsymbol = tmp + " " + num;                           // 拼接完整代码
    return fullsymbol;                                      // 返回完整代码
}
```

**逐行解释：**
- 第159-160行：创建两个字符串副本
- 第161行：从 `num` 中移除所有字母，只保留数字部分（合约月份）
- 第162行：从 `alpha` 中移除所有数字，只保留字母部分（合约ID）
- 第163行：使用字母部分从 `instrument2sec` 映射中查找对应的完整代码前缀
- 第164行：将完整代码前缀和数字部分拼接，中间加空格
- 第165行：返回完整代码

```cpp
    static char buf[1024 * 2];           // 静态缓冲区（2KB）
    string tmp = nowMS();                 // 获取当前时间（毫秒精度）
    size_t sz = tmp.size();
    strcpy(buf, tmp.c_str());            // 复制时间戳到缓冲区
    buf[sz] = ' ';                        // 添加空格分隔符

    va_list args;                         // 可变参数列表
    va_start(args, format);               // 初始化可变参数
    vsnprintf(buf + sz + 1, 1024 * 2 - sz - 1, format, args);  // 格式化字符串
    size_t buflen = strlen(buf);
    fwrite(buf, sizeof(char), buflen, logfile);  // 写入文件
    va_end(args);                         // 清理可变参数
}
```

**逐行解释：**
- 第80行：加锁保护，确保多线程安全
- 第82行：使用静态缓冲区避免频繁分配内存
- 第83行：获取当前时间戳（毫秒精度）

```cpp
CMsgqNanomsg::CMsgqNanomsg(MSGQ_PROTOCOL protocol, string url, bool binding)
    : CMsgq(protocol, url) {
    endpoint_ = url;
    
    // 根据协议类型创建不同的套接字
    if (protocol_ == MSGQ_PROTOCOL::PAIR) {
        sock_ = nn_socket(AF_SP, NN_PAIR);  // 创建PAIR套接字
        assert(sock_ >= 0);
        int32_t to = 100;  // 接收超时100毫秒
        assert(nn_setsockopt(sock_, NN_SOL_SOCKET, NN_RCVTIMEO, &to, sizeof(to)) >= 0);
        
        if (binding) {
            eid_ = nn_bind(sock_, endpoint_.c_str());      // 绑定地址
        } else {
            eid_ = nn_connect(sock_, endpoint_.c_str());   // 连接地址
        }
    } else if (protocol_ == MSGQ_PROTOCOL::PUB) {
        sock_ = nn_socket(AF_SP, NN_PUB);  // 创建PUB套接字
        assert(sock_ >= 0);
        int32_t sndbuff = 1024*1024*256;  // 发送缓冲区256MB
        assert(nn_setsockopt(sock_, NN_SOL_SOCKET, NN_SNDBUF, &sndbuff, sizeof(sndbuff)) >=0 );
        eid_ = nn_bind(sock_, endpoint_.c_str());  // PUB必须绑定
    } else if (protocol_ == MSGQ_PROTOCOL::SUB) {
        sock_ = nn_socket(AF_SP, NN_SUB);  // 创建SUB套接字
        assert(sock_ >= 0);
        int32_t to = 100;  // 接收超时100毫秒
        assert(nn_setsockopt(sock_, NN_SOL_SOCKET, NN_RCVTIMEO, &to, sizeof(to)) >= 0);
        nn_setsockopt(sock_, NN_SUB, NN_SUB_SUBSCRIBE, "", 0);  // 订阅所有主题
        eid_ = nn_connect(sock_, endpoint_.c_str());  // SUB必须连接
    } else if (protocol_ == MSGQ_PROTOCOL::PUSH) {
        sock_ = nn_socket(AF_SP, NN_PUSH);  // 创建PUSH套接字
        assert(sock_ >= 0);
        eid_ = nn_connect(sock_, endpoint_.c_str());  // PUSH连接
    } else if (protocol_ == MSGQ_PROTOCOL::PULL) {
        sock_ = nn_socket(AF_SP, NN_PULL);  // 创建PULL套接字
        assert(sock_ >= 0);
        eid_ = nn_bind(sock_, endpoint_.c_str());  // PULL绑定
    }

    if (eid_ < 0 || sock_ < 0) {
        LOG_ERROR(logger, "Nanomsg connect sock " << endpoint_ << "error");
    }
}
```

```cpp
        return nullptr;
    
    try {
        string des, src, stype;

        getline(ss, des, SERIALIZATION_SEPARATOR);   // 提取目标地址
        getline(ss, src, SERIALIZATION_SEPARATOR);   // 提取源地址

        switch (mtype) {
            case MSG_TYPE_ORDER:
                pheader = make_shared<OrderMsg>();
                pheader->msgtype_ = MSG_TYPE_ORDER;
                pheader->deserialize(msgin);
                break;
            case MSG_TYPE_ORDER_CTP:
                pheader = make_shared<CtpOrderMsg>();
                pheader->deserialize(msgin);
                break;
            case MSG_TYPE_ORDER_PAPER:
                pheader = make_shared<PaperOrderMsg>();
                pheader->deserialize(msgin);
                break;
            case MSG_TYPE_SUBSCRIBE_MARKET_DATA:
                pheader = make_shared<SubscribeMsg>(des, src);
                pheader->deserialize(msgin);
                break;
            case MSG_TYPE_UNSUBSCRIBE:
                pheader = make_shared<UnSubscribeMsg>(des, src);
                pheader->deserialize(msgin);
                break;
            case MSG_TYPE_ORDER_ACTION:
            case MSG_TYPE_CANCEL_ORDER:
                pheader = make_shared<OrderActionMsg>();
                pheader->deserialize(msgin);

            case MSG_TYPE_CANCEL_ALL:
                pheader = make_shared<CancelAllMsg>();
                pheader->deserialize(msgin);
                break;
            case MSG_TYPE_QRY_CONTRACT:
                pheader = make_shared<QryContractMsg>(des, src);
```

```cpp
    string msgpull = msgq_recv_->recmsg(0);  // 从PULL接收消息（非阻塞）
    if (msgpull.empty())
        return;
    
    // 特殊标志，表明消息让策略进程收到
    if (msgpull[0] == RELAY_DESTINATION) {  // '@'字符表示需要中继回策略

        // 将消息发回，让策略进程收到

        // 转发消息到各个engine

    }
}
```

### 5.8 时间处理函数

#### getMicroTime函数（util.cpp 第198-268行）

```cpp
uint64_t getMicroTime() {
    uint64_t t = 0;

#if defined(_WIN32) || defined(_WIN64)
    FILETIME tm;
#if defined(NTDDI_WIN8) && NTDDI_VERSION >= NTDDI_WIN8
    GetSystemTimePreciseAsFileTime(&tm);  // Windows 8+高精度时间
#else
    GetSystemTimeAsFileTime(&tm);         // Windows 2000+
#endif
    t = ((uint64_t)tm.dwHighDateTime << 32) | (uint64_t)tm.dwLowDateTime;
    return t;
#elif defined(__MACH__) && defined(__APPLE__)
    // macOS实现
    static double timeConvert = 0.0;
    if (timeConvert == 0.0) {
        mach_timebase_info_data_t timeBase;
        (void)mach_timebase_info(&timeBase);
        timeConvert = (double)timeBase.numer / (double)timeBase.denom / 1000000000.0;
    }
    return (double)mach_absolute_time() * timeConvert;
#elif defined(_POSIX_VERSION)
    // POSIX实现（Linux等）
    struct timespec ts;
    const clockid_t id = CLOCK_MONOTONIC;  // 单调时钟（不受系统时间调整影响）
    if (clock_gettime(id, &ts) != -1) {
        t = ts.tv_sec*__NANOO_MULTIPLE__ + ts.tv_nsec;
        return t / 1000;  // 转换为微秒
    }
    // 降级方案
    struct timeval tm;
    gettimeofday(&tm, nullptr);
    t = __MICRO_MULTIPLE__*tm.tv_sec + tm.tv_usec;
    return t;
#else
    return 0;
#endif
}
```

**说明：** 获取当前时间的微秒数，跨平台实现。

```cpp
    std::chrono::system_clock::duration tp = now.time_since_epoch();
    tp -= std::chrono::duration_cast<std::chrono::seconds>(tp);  // 提取秒以下部分

}
```
